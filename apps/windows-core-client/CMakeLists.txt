cmake_minimum_required(VERSION 3.20)
project(ConnectCoreClient LANGUAGES CXX)
add_executable(ConnectCore WIN32 main.cpp app.rc)
add_executable(ConnectCoreNetwork WIN32 network.cpp app.rc)
target_compile_features(ConnectCoreNetwork PRIVATE cxx_std_20)
target_compile_definitions(ConnectCoreNetwork PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(ConnectCoreNetwork PRIVATE fwpuclnt iphlpapi ws2_32 ole32 shell32 advapi32 wintrust crypt32 setupapi uuid)
target_compile_features(ConnectCore PRIVATE cxx_std_20)
target_compile_definitions(ConnectCore PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(ConnectCore PRIVATE d2d1 dwrite dwmapi comdlg32 shell32 ole32)
if(MSVC)
 target_compile_options(ConnectCore PRIVATE /W4 /utf-8 /EHsc)
 # app.rc embeds the DPI-aware asInvoker manifest; do not generate a second one.
 target_link_options(ConnectCore PRIVATE /MANIFEST:NO)
 set_property(TARGET ConnectCore PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
 target_compile_options(ConnectCoreNetwork PRIVATE /W4 /utf-8 /EHsc)
 target_link_options(ConnectCoreNetwork PRIVATE /MANIFEST:NO)
 set_property(TARGET ConnectCoreNetwork PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
